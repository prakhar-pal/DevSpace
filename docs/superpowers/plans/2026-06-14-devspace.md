# DevSpace Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Tauri desktop app that lists DevPod workspaces with live status, runs start/stop/delete/SSH actions, embeds a terminal for running workspaces, and routes Claude Code notifications from inside a pod to native desktop notifications.

**Architecture:** Rust (Tauri 2) backend shells out to the `devpod` CLI and manages PTY sessions + a local HTTP listener; a React/TypeScript frontend renders a sidebar of workspaces and an xterm.js terminal. Claude Code hooks inside a pod reach the desktop over a reverse SSH tunnel (`devpod ssh -R`).

**Tech Stack:** Rust, Tauri 2, `portable-pty`, `tauri-plugin-notification`, `tiny_http`, `serde`/`serde_json`; React + TypeScript + Vite, `@xterm/xterm`, `@tauri-apps/api`; Vitest.

**Reference:** Spec at `docs/superpowers/specs/2026-06-14-devspace-design.md`.

**Conventions for every Rust task:** run `cargo test` from `src-tauri/`. **Conventions for every frontend task:** run `npm test` (Vitest) and `npm run build` from repo root.

pub trait CommandRunner: Send + Sync {
    fn run(&self, args: &[&str]) -> Result<String, String>;
}

pub struct DevpodCli;

impl CommandRunner for DevpodCli {
    fn run(&self, args: &[&str]) -> Result<String, String> {
        let out = std::process::Command::new("devpod")
            .args(args)
            .output()
            .map_err(|e| format!("failed to spawn devpod: {e}"))?;
        if !out.status.success() {
            return Err(String::from_utf8_lossy(&out.stderr).trim().to_string());
        }
        Ok(String::from_utf8_lossy(&out.stdout).into_owned())
    }
}

pub fn list(runner: &dyn CommandRunner) -> Result<Vec<Workspace>, String> {
    let out = runner.run(&["list", "--output", "json"])?;
    parse_list(&out)
}

pub fn status(runner: &dyn CommandRunner, id: &str) -> Result<WorkspaceStatus, String> {
    let out = runner.run(&["status", id, "--output", "json", "--timeout", "10s"])?;
    parse_status(&out)
}

pub fn up(runner: &dyn CommandRunner, id: &str) -> Result<(), String> {
    runner.run(&["up", id, "--ide", "none"]).map(|_| ())
}

pub fn stop(runner: &dyn CommandRunner, id: &str) -> Result<(), String> {
    runner.run(&["stop", id]).map(|_| ())
}

pub fn delete(runner: &dyn CommandRunner, id: &str) -> Result<(), String> {
    runner.run(&["delete", id]).map(|_| ())
}

use serde::Serialize;
use std::collections::HashMap;

#[derive(Debug, Clone, Serialize)]
pub struct WorkspaceView {
    pub id: String,
    pub provider: String,
    pub source: String,
    pub last_used: String,
    pub state: String,
    pub error: Option<String>,
}

/// Merge the workspace list with a map of id -> status (or error string).
pub fn merge(
    workspaces: &[Workspace],
    statuses: &HashMap<String, Result<WorkspaceStatus, String>>,
) -> Vec<WorkspaceView> {
    workspaces
        .iter()
        .map(|w| {
            let source = w
                .source
                .git_repository
                .clone()
                .or_else(|| w.source.local_folder.clone())
                .unwrap_or_default();
            let (state, error) = match statuses.get(&w.id) {
                Some(Ok(s)) => (s.state.clone(), None),
                Some(Err(e)) => ("Unknown".to_string(), Some(e.clone())),
                None => ("Unknown".to_string(), None),
            };
            WorkspaceView {
                id: w.id.clone(),
                provider: w.provider.name.clone(),
                source,
                last_used: w.last_used.clone(),
                state,
                error,
            }
        })
        .collect()
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::devpod::parse_list;

    #[test]
    fn merge_applies_status_and_error() {
        let ws = parse_list(include_str!("../tests/fixtures/list.json")).unwrap();

        statuses.insert(

use crate::devpod::{self, CommandRunner};
use std::sync::Arc;

/// Run one poll cycle: list workspaces, fetch each status (capped concurrency
/// is unnecessary here because the runner is synchronous; statuses run
/// sequentially with the CLI's own short timeout), and return merged views.
pub fn poll_once(runner: &dyn CommandRunner) -> Result<Vec<WorkspaceView>, String> {
    let workspaces = devpod::list(runner)?;
    let mut statuses = HashMap::new();
    for w in &workspaces {
        statuses.insert(w.id.clone(), devpod::status(runner, &w.id));
    }
    Ok(merge(&workspaces, &statuses))
}

/// Spawn the background poll loop; emits `workspaces-updated` every `interval`.
pub fn spawn_loop<F>(runner: Arc<dyn CommandRunner>, interval: std::time::Duration, emit: F)
where
    F: Fn(Vec<WorkspaceView>) + Send + 'static,
{
    std::thread::spawn(move || loop {
        match poll_once(runner.as_ref()) {
            Ok(views) => emit(views),
            Err(_e) => { /* keep looping; transient CLI errors are expected */ }
        }
        std::thread::sleep(interval);
    });
}
```

import type { WorkspaceView } from "./api";

export function statusColor(state: string): string {
  switch (state) {
    case "Running":
      return "green";
    case "Stopped":
      return "grey";
    case "Busy":
      return "amber";
    default:
      return "red";
  }
}

interface Props {
  workspaces: WorkspaceView[];
  selectedId: string | null;
  badges: Record<string, boolean>;
  onSelect: (id: string) => void;
}

export function Sidebar({ workspaces, selectedId, badges, onSelect }: Props) {
  return (
    <nav className="sidebar">
      {workspaces.map((w) => (
        <button
          key={w.id}
          className={`ws-row${w.id === selectedId ? " selected" : ""}`}
          onClick={() => onSelect(w.id)}
          title={w.error ?? w.source}
        >
          <span className="dot" style={{ background: statusColor(w.state) }} />
          <span className="ws-name">{w.id}</span>
          {badges[w.id] && <span className="badge" aria-label="unread" />}
        </button>
      ))}
    </nav>
  );
}

pub const LISTENER_PORT: u16 = 17865;

/// Start a blocking HTTP listener on `LISTENER_PORT`. For each `POST /notify`
/// with a JSON `NotifyPayload`, invoke `on_notify(pod_id, message)`.
/// Returns Err if the port cannot be bound (caller logs and disables notifications).
pub fn serve<F>(on_notify: F) -> Result<(), String>
where
    F: Fn(String, String) + Send + 'static,
{
    let server = tiny_http::Server::http(("127.0.0.1", LISTENER_PORT))
        .map_err(|e| format!("notify listener bind failed: {e}"))?;
    std::thread::spawn(move || {
        for mut req in server.incoming_requests() {
            let mut body = String::new();
            use std::io::Read;
            let _ = req.as_reader().read_to_string(&mut body);
            if let Ok(p) = serde_json::from_str::<NotifyPayload>(&body) {
                on_notify(p.pod_id.clone(), message_for(&p.event).to_string());
            }
            let _ = req.respond(tiny_http::Response::empty(204));
        }
    });
    Ok(())
}

/// Merge the DevSpace notification hook into an existing settings document
/// (or an empty one), idempotently for the Notification, Stop, and
/// PreToolUse events. Existing unrelated hooks are preserved.
pub fn merge_settings(existing: Option<Value>) -> Value {
    let mut root = existing.unwrap_or_else(|| json!({}));
    if !root.is_object() {
        root = json!({});
    }
    let obj = root.as_object_mut().unwrap();
    let hooks = obj
        .entry("hooks")
        .or_insert_with(|| json!({}))
        .as_object_mut()
        .expect("hooks must be an object");

    for event in ["Notification", "Stop", "PreToolUse"] {
        let arr = hooks
            .entry(event)
            .or_insert_with(|| json!([]))
            .as_array_mut()
            .expect("event hooks must be an array");
        let cmd = hook_command();
        let already = arr.iter().any(|e| {
            e.get("hooks")
                .and_then(|h| h.as_array())
                .map(|h| h.iter().any(|x| x.get("command").and_then(|c| c.as_str()) == Some(cmd.as_str())))
                .unwrap_or(false)
        });
        if !already {
            arr.push(devspace_hook_entry());
        }
    }

use crate::devpod::CommandRunner;

/// Read the pod's ~/.claude/settings.json, merge in the DevSpace hooks, and
/// write it back via `devpod ssh --command`. Idempotent.
pub fn inject(runner: &dyn CommandRunner, pod_id: &str) -> Result<(), String> {
    let read = runner.run(&[
        "ssh", pod_id, "--command",
        "cat ~/.claude/settings.json 2>/dev/null || echo '{}'",
    ])?;
    let existing: Option<Value> = serde_json::from_str(read.trim()).ok();
    let merged = merge_settings(existing);
    let serialized = serde_json::to_string(&merged).map_err(|e| e.to_string())?;
    // base64 to safely transport JSON through the shell.
    let b64 = base64_encode(serialized.as_bytes());
    let cmd = format!(
        "mkdir -p ~/.claude && echo {b64} | base64 -d > ~/.claude/settings.json"
    );
    runner.run(&["ssh", pod_id, "--command", &cmd]).map(|_| ())
}

fn base64_encode(input: &[u8]) -> String {
    const T: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    let mut out = String::new();
    for chunk in input.chunks(3) {
        let b = [chunk[0], *chunk.get(1).unwrap_or(&0), *chunk.get(2).unwrap_or(&0)];
        out.push(T[(b[0] >> 2) as usize] as char);
        out.push(T[(((b[0] & 0x03) << 4) | (b[1] >> 4)) as usize] as char);
        out.push(if chunk.len() > 1 { T[(((b[1] & 0x0f) << 2) | (b[2] >> 6)) as usize] as char } else { '=' });
        out.push(if chunk.len() > 2 { T[(b[2] & 0x3f) as usize] as char } else { '=' });
    }
    out
}

use portable_pty::{CommandBuilder, NativePtySystem, PtyPair, PtySize, PtySystem};
use std::collections::HashMap;
use std::io::{Read, Write};
use std::sync::{Arc, Mutex};

pub struct Session {
    pair: PtyPair,
    writer: Box<dyn Write + Send>,
}

#[derive(Default)]
pub struct SessionRegistry {
    sessions: Mutex<HashMap<String, Arc<Mutex<Session>>>>,
}

impl SessionRegistry {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn has(&self, pod_id: &str) -> bool {
        self.sessions.lock().unwrap().contains_key(pod_id)
    }

    /// Spawn `devpod ssh ...` in a PTY. `on_output` is called with raw bytes as
    /// they arrive from the workspace. The session persists until `close`.
    pub fn ensure<F>(&self, pod_id: &str, on_output: F) -> Result<(), String>
    where
        F: Fn(Vec<u8>) + Send + 'static,
    {
        let mut map = self.sessions.lock().unwrap();
        if map.contains_key(pod_id) {
            return Ok(());
        }
        let pty = NativePtySystem::default();
        let pair = pty
            .openpty(PtySize { rows: 24, cols: 80, pixel_width: 0, pixel_height: 0 })
            .map_err(|e| e.to_string())?;

        let mut cmd = CommandBuilder::new("devpod");
        for a in ssh_args(pod_id) {
            cmd.arg(a);
        }
        let _child = pair.slave.spawn_command(cmd).map_err(|e| e.to_string())?;

        let mut reader = pair.master.try_clone_reader().map_err(|e| e.to_string())?;
        std::thread::spawn(move || {
            let mut buf = [0u8; 4096];
            loop {
                match reader.read(&mut buf) {
                    Ok(0) | Err(_) => break,
                    Ok(n) => on_output(buf[..n].to_vec()),
                }
            }
        });

        let writer = pair.master.take_writer().map_err(|e| e.to_string())?;
        map.insert(pod_id.to_string(), Arc::new(Mutex::new(Session { pair, writer })));
        Ok(())
    }

    pub fn write(&self, pod_id: &str, data: &[u8]) -> Result<(), String> {
        let map = self.sessions.lock().unwrap();
        let sess = map.get(pod_id).ok_or("no session")?.clone();
        let mut s = sess.lock().unwrap();
        s.writer.write_all(data).map_err(|e| e.to_string())?;
        s.writer.flush().map_err(|e| e.to_string())
    }

    pub fn resize(&self, pod_id: &str, rows: u16, cols: u16) -> Result<(), String> {
        let map = self.sessions.lock().unwrap();
        let sess = map.get(pod_id).ok_or("no session")?.clone();
        let s = sess.lock().unwrap();
        s.pair
            .master
            .resize(PtySize { rows, cols, pixel_width: 0, pixel_height: 0 })
            .map_err(|e| e.to_string())
    }

    pub fn close(&self, pod_id: &str) {
        self.sessions.lock().unwrap().remove(pod_id);
    }
}
```

import { Terminal as XTerm } from "@xterm/xterm";
import { FitAddon } from "@xterm/addon-fit";
import "@xterm/xterm/css/xterm.css";
import { connectPty, writePty, resizePty, onPtyOutput } from "./api";

export function Terminal({ podId }: { podId: string }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!ref.current) return;
    const term = new XTerm({ convertEol: true, fontSize: 13 });
    const fit = new FitAddon();
    term.loadAddon(fit);
    term.open(ref.current);
    fit.fit();

    let unlisten: (() => void) | undefined;
    const enc = new TextEncoder();

    (async () => {
      unlisten = await onPtyOutput(podId, (bytes) =>
        term.write(new Uint8Array(bytes))
      );
      await connectPty(podId);
      await resizePty(podId, term.rows, term.cols);
    })();

    const onData = term.onData((data) =>
      writePty(podId, Array.from(enc.encode(data)))
    );
    const onResize = term.onResize(({ rows, cols }) =>
      resizePty(podId, rows, cols)
    );
    const handleWindowResize = () => fit.fit();
    window.addEventListener("resize", handleWindowResize);

    return () => {
      window.removeEventListener("resize", handleWindowResize);
      onData.dispose();
      onResize.dispose();
      unlisten?.();
      term.dispose();
    };
  }, [podId]);

  return <div className="terminal" ref={ref} />;
}
